function confirmExit() { if (totalWrongSet.size > 0 && confirm(`오답 리스트(${totalWrongSet.size}건)를 엑셀 형식으로 다운로드할까요?`)) { // UTF-8 BOM으로 엑셀 한글 깨짐 방지 let csv = "\uFEFF"; // 헤더 설정 (요청하신 형식) csv += "구분,번호,한자,병음,한글 뜻\n"; totalWrongSet.forEach((item, index) => { const no = index + 1; // 질문 행 csv += `질문번호,${no},"${item.q}","${item.qP}","${item.hint || ''}"\n`; // 답변 행 csv += `답변번호,${no},"${item.a}","${item.aP}","(답변 뜻)"\n`; // 가독성을 위한 빈 줄 (선택 사항) csv += ",,,,\n"; }); const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' }); const a = document.createElement("a"); const url = URL.createObjectURL(blob); a.href = url; a.download = `TSC_오답노트_${new Date().toLocaleDateString()}.csv`; a.click(); URL.revokeObjectURL(url); } location.reload(); }
